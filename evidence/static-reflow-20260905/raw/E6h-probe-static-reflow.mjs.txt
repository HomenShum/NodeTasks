import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { createServer } from 'node:http';
import { join, dirname } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';

const { chromium } = await import(pathToFileURL('D:/VSCode Projects/cafecorner_nodebench/nodebench_ai4/.portfolio-worktrees/nodevideo-current-consumer-20260905/node_modules/playwright/index.mjs').href);

const packet = dirname(fileURLToPath(import.meta.url));
const root = 'D:/VSCode Projects/cafecorner_nodebench/nodebench_ai4/.portfolio-worktrees/nodetasks-corpus-readiness-20260905';
const out = join(packet, 'E6h-nodetasks-static-causal-before-01');
await mkdir(out);
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const names = ['scripts/build-catalog.mjs', 'catalog/task-browser.html', 'catalog/search-index.js'];
const before = await Promise.all(names.map(async path => ({path, sha256:hash(await readFile(join(root,path)))})));
const payloads = new Map();
for (const name of ['task-browser.html','search-index.js']) payloads.set(`/${name}`,await readFile(join(root,'catalog',name)));
const requests=[];
const server=createServer((request,response)=>{
  requests.push({method:request.method,path:request.url});
  if (requests.length>100) {response.writeHead(429);response.end();return;}
  const bytes=payloads.get(request.url);
  if (!bytes || request.method!=='GET') {response.writeHead(404);response.end();return;}
  response.writeHead(200,{'content-type':request.url.endsWith('.js')?'text/javascript; charset=utf-8':'text/html; charset=utf-8','content-length':bytes.length});response.end(bytes);
});
await new Promise(done=>server.listen(0,'127.0.0.1',done));
const port=server.address().port;
const base=`http://127.0.0.1:${port}`;
const browser=await chromium.launch();
const report={status:'RUNNING',base,viewport:[390,844],sourceBefore:before,requests,errors:[],captures:[],scope:'Read-only current source plus reversible in-browser CSS causal probe; no product file edit, no grade.'};
try {
  const context=await browser.newContext({viewport:{width:390,height:844},reducedMotion:'no-preference'});
  await context.route('**/*',route=>new URL(route.request().url()).origin===base ? route.continue():route.abort('blockedbyclient'));
  const page=await context.newPage();page.setDefaultTimeout(15000);
  page.on('pageerror',error=>report.errors.push(String(error)));
  await page.goto(`${base}/task-browser.html`);
  await page.locator('#query').fill('nodeagent.storyboard.capture.v1');
  assert.equal(await page.locator('#summary').innerText(),'1 shown');
  const capture=async name=>{
    const observation=await page.evaluate(()=>({
      scrollX:window.scrollX,viewport:[innerWidth,innerHeight],documentWidth:document.documentElement.scrollWidth,
      elements:[...document.querySelectorAll('body,.shell,aside,main,.bar,.result,.result code')].map(el=>{
        const s=getComputedStyle(el),r=el.getBoundingClientRect();
        return {selector:el.className||el.tagName,text:el.textContent.trim().slice(0,150),x:r.x,y:r.y,width:r.width,right:r.right,minWidth:s.minWidth,gridTemplateColumns:s.gridTemplateColumns,overflowWrap:s.overflowWrap,flexWrap:s.flexWrap,overflowX:s.overflowX};
      }),resultText:document.querySelector('.result').textContent,
    }));
    await page.screenshot({path:join(out,`${name}.png`)});
    await writeFile(join(out,`${name}.html`),await page.content());
    await writeFile(join(out,`${name}.json`),JSON.stringify(observation,null,2));
    report.captures.push({name,...observation});return observation;
  };
  const original=await capture('before');assert.equal(original.documentWidth-390,200);
  const css='.shell { grid-template-columns: 280px minmax(0, 1fr); } main { min-width: 0; } .result { overflow-wrap: anywhere; } .bar { flex-wrap: wrap; } @media (max-width: 820px) { .shell { grid-template-columns: minmax(0, 1fr); } }';
  await writeFile(join(out,'probe.css.txt'),css);
  const style=await page.addStyleTag({content:css});await page.evaluate(()=>window.scrollTo(0,0));
  const probe=await capture('probe');assert.equal(probe.documentWidth,390);assert.equal(probe.resultText,original.resultText);
  await style.evaluate(el=>el.remove());await page.locator('#query').focus();
  const restored=await capture('restored');assert.equal(restored.documentWidth-390,200);assert.equal(restored.resultText,original.resultText);
  report.status='CAUSAL_PROBE_PASS_SOURCE_UNCHANGED';report.productFilesChanged=0;
  await context.close();
} catch(error) {report.status='FAIL';report.error=String(error);process.exitCode=1;}
finally {
  await browser.close();await new Promise(done=>server.close(done));
  for(const entry of before) assert.equal(hash(await readFile(join(root,entry.path))),entry.sha256);
  report.sourceUnchanged=true;report.runtimeClosed=true;report.finishedAt=new Date().toISOString();
  await writeFile(join(out,'report.json'),JSON.stringify(report,null,2));
  console.log(JSON.stringify({status:report.status,sourceUnchanged:report.sourceUnchanged,runtimeClosed:report.runtimeClosed,out}));
}
