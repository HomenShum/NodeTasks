def check(name,c,detail=None):
 checks.append({'name':name,'passed':bool(c),'detail':detail})
 if not c:raise AssertionError(name)

def run(name,args,stdin=None):
 result=subprocess.run(args,cwd=R,env=env,input=stdin,capture_output=True,timeout=60)
 (O/(name+'.stdout')).write_bytes(result.stdout);(O/(name+'.stderr')).write_bytes(result.stderr)
 return result

  b=(Q/row['path']).read_bytes();check('Manifest raw identity '+row['path'],len(b)==row['bytes'] and sha(b)==row['sha256'] and hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest()==row['gitBlob'])
