sha = lambda b: hashlib.sha256(b).hexdigest()
blob = lambda b: hashlib.sha1(f'blob {len(b)}\0'.encode() + b).hexdigest()

def dump(name, value):
    (O/name).write_text(json.dumps(value, ensure_ascii=False, indent=2)+'\n', encoding='utf-8', newline='\n')
def bind(p, name):
    b=p.read_bytes()
    return dict(path=name, bytes=len(b), sha256=sha(b), gitBlobSha1=blob(b))

archive_rows=[]; changed=[]; diffs=[]

    for member in tf.getmembers():

        b=tf.extractfile(member).read()

        if not rel: continue
        archive_rows.append(dict(path=rel,bytes=len(b),sha256=sha(b),gitBlobSha1=blob(b)))
        current=(R/rel).read_bytes()
        if current!=b: changed.append(rel)
