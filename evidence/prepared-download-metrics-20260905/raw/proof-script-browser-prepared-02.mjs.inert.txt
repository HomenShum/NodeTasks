async function committedInput(page, action, renderedOutcome) {
  await settled(page);
  await page.evaluate(() => {
    const app = document.querySelector('[data-testid="stApp"]');
    window.__nodetasksObservedRun = { states: [] };
    const record = window.__nodetasksObservedRun;
    record.observer = new MutationObserver(() => {
      const state = app.getAttribute("data-test-script-state");
      if (record.states.length < 16 && record.states.at(-1) !== state) record.states.push(state);
    });
    record.observer.observe(app, { attributes: true, attributeFilter: ["data-test-script-state"] });
  });
  try {
    await action();
    // Fast cached reruns need not expose a sampled busy frame. Require the actual
    // updated table/metric, then inspect the exported bytes independently.
    await page.waitForFunction(renderedOutcome);
    await page.waitForFunction(() => document.querySelector('[data-testid="stApp"]').getAttribute("data-test-script-state") === "notRunning");
    await settled(page);
    return await page.evaluate(() => window.__nodetasksObservedRun.states);
  } finally {
    await page.evaluate(() => { window.__nodetasksObservedRun.observer.disconnect(); delete window.__nodetasksObservedRun; });
  }
}
async function capture(page, cell, name) {

}

async function download(page, button, cell, name, keyboard = false) {
  const event = page.waitForEvent("download");
  if (keyboard) await page.keyboard.press("Enter");
  else await button.click();
  const item = await event;
  assert.equal(await item.failure(), null);
  const path = join(output, `${name}.json`);
  await item.saveAs(path);
  const bytes = await readFile(path);
  const payload = JSON.parse(bytes);
  cell.downloads.push({ name, suggestedFilename: item.suggestedFilename(), sha256: hash(bytes), bytes: bytes.length });
  return payload;
}

async function preparedContext(page) {
  const detail = page.getByTestId("stExpander").filter({ hasText: "Prepared filters" });
  if (!(await detail.locator("details").evaluate((element) => element.open))) await detail.locator("summary").click();
  return detail.getByTestId("stJson").evaluate((element) => Object.fromEntries([...element.querySelectorAll(".object-key")].map((key) => {
    const row = key.closest(".variable-row, .object-key-val");
    const value = row.classList.contains("variable-row") ? row.querySelector(".variable-value") : null;
    return [JSON.parse(key.textContent), value ? JSON.parse(value.textContent) : [...row.querySelectorAll(".string-value")].map((item) => JSON.parse(item.textContent))];
  })));
}

async function metricFacts(page) {
  const facts = await page.getByTestId("stMetric").evaluateAll((elements) => elements.map((element) => ({
    text: element.textContent,
    background: getComputedStyle(element).backgroundColor,
    label: [...element.querySelectorAll('[data-testid="stMetricLabel"], [data-testid="stMetricLabel"] *')].map((child) => ({ text: child.textContent, color: getComputedStyle(child).color })),
    value: [...element.querySelectorAll('[data-testid="stMetricValue"], [data-testid="stMetricValue"] *')].map((child) => ({ text: child.textContent, color: getComputedStyle(child).color })),
    delta: [...element.querySelectorAll('[data-testid="stMetricDelta"], [data-testid="stMetricDelta"] *')].map((child) => ({ tag: child.tagName, text: child.textContent, color: getComputedStyle(child).color, background: getComputedStyle(child).backgroundColor, fill: getComputedStyle(child).fill, path: child.getAttribute("d") })),
  })));
  assert.equal(facts.length, 13, "Five summary, five task and three provenance metrics");

    assert.equal(fact.background, "rgb(15, 20, 25)");
    for (const text of [...fact.label, ...fact.value]) assert.equal(text.color, "rgb(229, 231, 235)");
  }
  return facts;
}

  }
  report.status = report.cells.length === viewports.length && report.cells.every((cell) => cell.status === "OBSERVED") ? "OBSERVED_COMPLETE_WITH_UNGRADED_VISUALS" : "FAILED";
} catch (error) { report.status = "FAILED"; report.errors.push(String(error)); }
finally {
  if (browser) await browser.close();
  if (server.exitCode === null) spawnSync("taskkill", ["/PID", String(server.pid), "/T", "/F"], { windowsHide: true });
  await new Promise((done) => server.exitCode !== null ? done() : server.once("exit", done));
  await new Promise((done) => serverLog.end(done));
  await new Promise((done) => staticServer.close(done));
  report.sourceUnchanged = (await Promise.all(inputs.map(async (input) => hash(await readFile(join(root, input.path))) === input.sha256))).every(Boolean);
  assert.equal(report.sourceUnchanged, true);
  report.finishedAt = new Date().toISOString();
  await writeFile(join(output, "report.json"), JSON.stringify(report, null, 2));
}
console.log(JSON.stringify({ status: report.status, cells: report.cells.length, output }));
process.exitCode = report.status === "OBSERVED_COMPLETE_WITH_UNGRADED_VISUALS" ? 0 : 1;
